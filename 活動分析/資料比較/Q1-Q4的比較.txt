# -*- coding: utf-8 -*-
"""
長輩全年度（Q1 至 Q4）四季綜合對比分析系統（差異改為 Q4 - Q3 版）
- 自動至 115Q1活動分析 至 115Q4活動分析 抓取平均報表
- 橫向並列 Q1、Q2、Q3、Q4，右側「Q4 相比 Q3 差異 (Q4 - Q3)」欄位 (AA 到 AF)
- 總平均差距列填入 Q4 - Q3 帶有箭頭（▲綠 / ▼紅）的具體差值
- 自動存入「115Q1-Q4(全年度)」資料夾中
"""

import os
import re
import pandas as pd
import openpyxl
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter

target_activities = ['手部(大肢體)', '手部(精細)', '下肢', '上下肢全身', '認知', '【跨月份總平均】']
indicators = ['參與度', '自我功能', '情緒控制力', '與他人互動', '肢體表現', '參與場次']

if __name__ == '__main__':
    print('=' * 60)
    print('長輩全年度（Q1-Q4）四季綜合對比分析系統（Q4 - Q3 差異版）')
    print('=' * 60)

    output_dir = "115Q1-Q4(全年度)"
    os.makedirs(output_dir, exist_ok=True)

    source_folders = ['115Q1活動分析', '115Q2活動分析', '115Q3活動分析', '115Q4活動分析']
    all_file_paths = []

    for folder in source_folders:
        if os.path.exists(folder):
            sub_files = [
                os.path.join(folder, f)
                for f in os.listdir(folder)
                if f.endswith('.xlsx') and '平均報表_' in f and not f.startswith('~$')
            ]
            all_file_paths.extend(sub_files)
            print(f"--> 從資料夾 [{folder}] 找到 {len(sub_files)} 份報表")
        else:
            print(f"【提示】找不到來源資料夾「{folder}」，將略過。")

    if not all_file_paths:
        print('\n【錯誤】找不到任何已產出的「平均報表」檔案！')
        exit()

    elder_master_data = {}
    elder_cms = {}

    for file_path in all_file_paths:
        file_name = os.path.basename(file_path)
        m = re.match(r'(.+?)平均報表_(.+)\.xlsx', file_name)
        if not m:
            continue

        quarter_name = m.group(1).strip()
        if '01-03' in quarter_name or quarter_name == '11501-11503':
            q_label = "Q1 (11501-11503)"
        elif '04-06' in quarter_name or quarter_name == '11504-11506':
            q_label = "Q2 (11504-11506)"
        elif '07-09' in quarter_name or quarter_name == '11507-11509':
            q_label = "Q3 (11507-11509)"
        elif '10-12' in quarter_name or quarter_name == '11510-11512':
            q_label = "Q4 (11510-11512)"
        else:
            continue

        elder_name = m.group(2).strip()

        try:
            wb_temp = openpyxl.load_workbook(file_path, data_only=True)
            ws1 = wb_temp.worksheets[0]

            sub_text = str(ws1['B3'].value or '')
            m_cms = re.search(r'CMS\s*([0-9０-９]+)', sub_text)
            if m_cms and elder_name not in elder_cms:
                elder_cms[elder_name] = m_cms.group(1)

            df_summary = pd.read_excel(file_path, sheet_name=0, header=None)

            header_row_idx = None
            for r in range(len(df_summary)):
                val = str(df_summary.iloc[r, 1]).strip()
                if val == '活動類型':
                    header_row_idx = r
                    break

            if header_row_idx is not None:
                if elder_name not in elder_master_data:
                    elder_master_data[elder_name] = {}

                for r in range(header_row_idx + 1, len(df_summary)):
                    act_type = str(df_summary.iloc[r, 1]).strip()
                    if not act_type or act_type == 'nan':
                        continue

                    if act_type not in target_activities and '總平均' not in act_type and '跨月份總平均' not in act_type:
                        continue

                    if '總平均' in act_type or '跨月份總平均' in act_type:
                        act_type = '【跨月份總平均】'

                    if act_type not in elder_master_data[elder_name]:
                        elder_master_data[elder_name][act_type] = {}

                    row_scores = {}
                    orig_inds = ['參與度', '自我功能', '情緒控制力', '與他人互動', '肢體表現', '參與場次']
                    for c_idx, ind_name in enumerate(orig_inds):
                        val_score = df_summary.iloc[r, 2 + c_idx]
                        row_scores[indicators[c_idx]] = val_score if pd.notna(val_score) else 0.0

                    elder_master_data[elder_name][act_type][q_label] = row_scores

            print(f"  -> 成功載入 [{elder_name}] 的 [{q_label}] 數據")

        except Exception as e:
            print(f"  [讀取警告] {file_name}: {e}")

    print(f"\n" + "=" * 50)
    print(" 開始產出全年度四季對比報表（Q4 - Q3 差異與紅綠箭頭）...")
    print("=" * 50)

    navy_header = PatternFill(start_color="1F4E79", end_color="1F4E79", fill_type="solid")
    navy_accent = PatternFill(start_color="2F5597", end_color="2F5597", fill_type="solid")

    q2_header = PatternFill(start_color="3B3838", end_color="3B3838", fill_type="solid")
    q2_accent = PatternFill(start_color="595959", end_color="595959", fill_type="solid")

    q3_header = PatternFill(start_color="264653", end_color="264653", fill_type="solid")
    q3_accent = PatternFill(start_color="2A9D8F", end_color="2A9D8F", fill_type="solid")

    q4_header = PatternFill(start_color="4A3B5C", end_color="4A3B5C", fill_type="solid")
    q4_accent = PatternFill(start_color="7B68AE", end_color="7B68AE", fill_type="solid")

    diff_header = PatternFill(start_color="2F3E46", end_color="2F3E46", fill_type="solid")
    diff_accent = PatternFill(start_color="52796F", end_color="52796F", fill_type="solid")

    zebra_fill = PatternFill(start_color="F9FAFB", end_color="F9FAFB", fill_type="solid")
    trend_row_fill = PatternFill(start_color="EBF1F5", end_color="EBF1F5", fill_type="solid")

    white_font = Font(name="微軟正黑體", size=11, bold=True, color="FFFFFF")
    title_font = Font(name="微軟正黑體", size=15, bold=True, color="1F4E79")
    subtitle_font = Font(name="微軟正黑體", size=10, color="595959")
    regular_font = Font(name="微軟正黑體", size=10)
    bold_font = Font(name="微軟正黑體", size=10, bold=True)

    thin_border = Border(
        left=Side(style='thin', color='D9D9D9'), right=Side(style='thin', color='D9D9D9'),
        top=Side(style='thin', color='D9D9D9'), bottom=Side(style='thin', color='D9D9D9')
    )
    split_border = Border(
        left=Side(style='thin', color='D9D9D9'), right=Side(style='medium', color='1F4E79'),
        top=Side(style='thin', color='D9D9D9'), bottom=Side(style='thin', color='D9D9D9')
    )
    diff_split_border = Border(
        left=Side(style='thin', color='D9D9D9'), right=Side(style='medium', color='2F3E46'),
        top=Side(style='thin', color='D9D9D9'), bottom=Side(style='thin', color='D9D9D9')
    )

    for elder_name, act_data in elder_master_data.items():
        wb = openpyxl.Workbook()
        ws1 = wb.active
        ws1.title = "全年度四季對比總表"
        ws1.views.sheetView[0].showGridLines = True

        cms_val = elder_cms.get(elder_name, '')
        cms_str = f"CMS {cms_val} 級" if cms_val else "CMS (未填)"

        ws1['B2'] = f"長輩【{elder_name}】全年度 (Q1 - Q4) 活動指標四方橫向對比暨 Q4-Q3 差異總表"
        ws1['B2'].font = title_font
        ws1['B3'] = f"長照等級：{cms_str} ｜ 第一季至第四季指標追蹤與 Q4 相比 Q3 成長趨勢（▲綠 / ▼紅）"
        ws1['B3'].font = subtitle_font

        ws1.merge_cells('B5:B6')
        ws1.cell(row=5, column=2, value="活動類型").fill = navy_header
        ws1.cell(row=5, column=2).font = white_font
        ws1.cell(row=5, column=2).alignment = Alignment(horizontal="center", vertical="center")

        ws1.merge_cells('C5:H5')
        ws1.cell(row=5, column=3, value="Q1 (11501-11503)").fill = navy_header
        ws1.cell(row=5, column=3).font = white_font
        ws1.cell(row=5, column=3).alignment = Alignment(horizontal="center", vertical="center")

        ws1.merge_cells('I5:N5')
        ws1.cell(row=5, column=9, value="Q2 (11504-11506)").fill = q2_header
        ws1.cell(row=5, column=9).font = white_font
        ws1.cell(row=5, column=9).alignment = Alignment(horizontal="center", vertical="center")

        ws1.merge_cells('O5:T5')
        ws1.cell(row=5, column=15, value="Q3 (11507-11509)").fill = q3_header
        ws1.cell(row=5, column=15).font = white_font
        ws1.cell(row=5, column=15).alignment = Alignment(horizontal="center", vertical="center")

        ws1.merge_cells('U5:Z5')
        ws1.cell(row=5, column=21, value="Q4 (11510-11512)").fill = q4_header
        ws1.cell(row=5, column=21).font = white_font
        ws1.cell(row=5, column=21).alignment = Alignment(horizontal="center", vertical="center")

        ws1.merge_cells('AA5:AF5')
        ws1.cell(row=5, column=27, value="Q4 相比 Q3 差異 (Q4 - Q3)").fill = diff_header
        ws1.cell(row=5, column=27).font = white_font
        ws1.cell(row=5, column=27).alignment = Alignment(horizontal="center", vertical="center")

        for c_idx in range(3, 9):
            cell = ws1.cell(row=6, column=c_idx, value=indicators[(c_idx - 3) % 6])
            cell.fill = navy_accent; cell.font = white_font; cell.alignment = Alignment(horizontal="center", vertical="center")

        for c_idx in range(9, 15):
            cell = ws1.cell(row=6, column=c_idx, value=indicators[(c_idx - 9) % 6])
            cell.fill = q2_accent; cell.font = white_font; cell.alignment = Alignment(horizontal="center", vertical="center")

        for c_idx in range(15, 21):
            cell = ws1.cell(row=6, column=c_idx, value=indicators[(c_idx - 15) % 6])
            cell.fill = q3_accent; cell.font = white_font; cell.alignment = Alignment(horizontal="center", vertical="center")

        for c_idx in range(21, 27):
            cell = ws1.cell(row=6, column=c_idx, value=indicators[(c_idx - 21) % 6])
            cell.fill = q4_accent; cell.font = white_font; cell.alignment = Alignment(horizontal="center", vertical="center")

        for c_idx in range(27, 33):
            cell = ws1.cell(row=6, column=c_idx, value=indicators[(c_idx - 27) % 6])
            cell.fill = diff_accent; cell.font = white_font; cell.alignment = Alignment(horizontal="center", vertical="center")

        ws1.row_dimensions[5].height = 22
        ws1.row_dimensions[6].height = 22
        ws1.cell(row=6, column=2).fill = navy_header

        current_row = 7
        row_idx_counter = 0

        for act_type in target_activities:
            if act_type not in act_data:
                continue

            ws1.cell(row=current_row, column=2, value=act_type).font = bold_font
            ws1.cell(row=current_row, column=2).alignment = Alignment(horizontal="left", vertical="center")
            ws1.cell(row=current_row, column=2).border = thin_border

            q1_scores = act_data[act_type].get("Q1 (11501-11503)", {})
            q2_scores = act_data[act_type].get("Q2 (11504-11506)", {})
            q3_scores = act_data[act_type].get("Q3 (11507-11509)", {})
            q4_scores = act_data[act_type].get("Q4 (11510-11512)", {})

            for idx, ind in enumerate(indicators):
                c_q1 = 3 + idx
                v_q1 = q1_scores.get(ind, 0.0)
                cell = ws1.cell(row=current_row, column=c_q1, value=v_q1)
                cell.font = regular_font; cell.alignment = Alignment(horizontal="center", vertical="center")
                cell.border = split_border if c_q1 == 8 else thin_border
                cell.number_format = '0' if ind == '參與場次' else '0.0'

                c_q2 = 9 + idx
                v_q2 = q2_scores.get(ind, 0.0)
                cell = ws1.cell(row=current_row, column=c_q2, value=v_q2)
                cell.font = regular_font; cell.alignment = Alignment(horizontal="center", vertical="center")
                cell.border = split_border if c_q2 == 14 else thin_border
                cell.number_format = '0' if ind == '參與場次' else '0.0'

                c_q3 = 15 + idx
                v_q3 = q3_scores.get(ind, 0.0)
                cell = ws1.cell(row=current_row, column=c_q3, value=v_q3)
                cell.font = regular_font; cell.alignment = Alignment(horizontal="center", vertical="center")
                cell.border = split_border if c_q3 == 20 else thin_border
                cell.number_format = '0' if ind == '參與場次' else '0.0'

                c_q4 = 21 + idx
                v_q4 = q4_scores.get(ind, 0.0)
                cell = ws1.cell(row=current_row, column=c_q4, value=v_q4)
                cell.font = regular_font; cell.alignment = Alignment(horizontal="center", vertical="center")
                cell.border = diff_split_border if c_q4 == 26 else thin_border
                cell.number_format = '0' if ind == '參與場次' else '0.0'

                # 計算 Q4 - Q3 差異
                col_diff = 27 + idx
                if ind == '參與場次':
                    diff_val = int(v_q4) - int(v_q3)
                else:
                    diff_val = round(float(v_q4) - float(v_q3), 1)

                if diff_val > 0:
                    diff_font = Font(name="微軟正黑體", size=10, bold=True, color="006100")
                elif diff_val < 0:
                    diff_font = Font(name="微軟正黑體", size=10, bold=True, color="9C0006")
                else:
                    diff_font = Font(name="微軟正黑體", size=10, bold=False, color="000000")

                cell_diff = ws1.cell(row=current_row, column=col_diff, value=diff_val)
                cell_diff.font = diff_font
                cell_diff.alignment = Alignment(horizontal="center", vertical="center")
                cell_diff.border = thin_border
                cell_diff.number_format = '0' if ind == '參與場次' else '+0.0;-0.0;0.0'

            if row_idx_counter % 2 == 1:
                for c in range(2, 33):
                    ws1.cell(row=current_row, column=c).fill = zebra_fill

            ws1.row_dimensions[current_row].height = 20
            current_row += 1
            row_idx_counter += 1

        # 總平均差距總結列（Q4 - Q3 帶 ▲綠 / ▼紅 箭頭）
        trend_row_idx = current_row
        ws1.cell(row=trend_row_idx, column=2, value="【總平均差距 (Q4-Q3)】").font = bold_font
        ws1.cell(row=trend_row_idx, column=2).alignment = Alignment(horizontal="left", vertical="center")
        ws1.cell(row=trend_row_idx, column=2).border = thin_border
        ws1.cell(row=trend_row_idx, column=2).fill = trend_row_fill

        for c in range(3, 27):
            cell = ws1.cell(row=trend_row_idx, column=c, value="")
            if c in (8, 14, 20):
                cell.border = split_border
            elif c == 26:
                cell.border = diff_split_border
            else:
                cell.border = thin_border
            cell.fill = trend_row_fill

        q3_total_scores = act_data.get('【跨月份總平均】', {}).get("Q3 (11507-11509)", {})
        q4_total_scores = act_data.get('【跨月份總平均】', {}).get("Q4 (11510-11512)", {})

        for idx, ind in enumerate(indicators):
            col_trend = 27 + idx
            v3 = q3_total_scores.get(ind, 0.0)
            v4 = q4_total_scores.get(ind, 0.0)

            if ind == '參與場次':
                d_val = int(v4) - int(v3)
            else:
                d_val = round(float(v4) - float(v3), 1)

            cell_trend = ws1.cell(row=trend_row_idx, column=col_trend)
            cell_trend.alignment = Alignment(horizontal="center", vertical="center")
            cell_trend.border = thin_border
            cell_trend.fill = trend_row_fill

            if d_val > 0:
                cell_trend.value = f"▲ +{d_val}"
                cell_trend.font = Font(name="微軟正黑體", size=10, bold=True, color="006100")
                cell_trend.number_format = '@'
            elif d_val < 0:
                cell_trend.value = f"▼ {d_val}"
                cell_trend.font = Font(name="微軟正黑體", size=10, bold=True, color="9C0006")
                cell_trend.number_format = '@'
            else:
                cell_trend.value = " 0.0" if ind != '參與場次' else " 0"
                cell_trend.font = Font(name="微軟正黑體", size=10, bold=False, color="000000")
                cell_trend.number_format = '@'

        ws1.row_dimensions[trend_row_idx].height = 22

        for col in ws1.columns:
            col_letter = get_column_letter(col[0].column)
            if col_letter == 'B':
                ws1.column_dimensions[col_letter].width = 24
            else:
                max_len = max(sum(2 if ord(char) > 127 else 1 for char in str(cell.value or '')) for cell in col)
                ws1.column_dimensions[col_letter].width = max(max_len + 3, 10)
        ws1.column_dimensions['A'].width = 4

        safe_name = re.sub(r'[\\/:*?"<>|]', '', elder_name).strip()
        output_file = os.path.join(output_dir, f'115全年度四季對比報表_{safe_name}.xlsx')

        wb.save(output_file)
        print(f"  [完成全年度報表] {elder_name} -> {output_file}")

    print('\n' + '=' * 60)
    print(" 所有長輩的全年度四季對比報表（含 Q4-Q3 紅綠箭頭）已產出，並存入 '115Q1-Q4(全年度)' 資料夾！")
    print('=' * 60)
